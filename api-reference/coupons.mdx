title: "Create Coupon"
openapi: "POST /api/v1/coupons"
description: "Create a new discount coupon."
---

<ParamField body="code" type="string" required>
  The unique coupon code. This code will be used by customers to redeem the discount.
</ParamField>

<ParamField body="description" type="string">
  A description of the coupon and its discount offer.
</ParamField>

<ParamField body="discount_kind" type="string" required>
  The type of discount applied by this coupon.
  Allowed values: `PERCENTAGE`.
</ParamField>

<ParamField body="discount_value" type="integer" required>
  The discount value. The meaning depends on `discount_kind`:
  
  - For `PERCENTAGE`: Discount percentage (1-100)
</ParamField>

<ParamField body="max_redeems" type="integer" required>
  Maximum number of times this coupon can be redeemed. Must be at least 1.
</ParamField>

<ParamField body="expires_at" type="string">
  Expiration date and time for the coupon in ISO 8601 format (e.g., "2024-12-31T23:59:59Z"). If not provided, the coupon does not expire.
</ParamField>

<ParamField body="is_active" type="boolean">
  Whether the coupon is currently active and can be used. Defaults to `true` if not provided.
</ParamField>

interface CreateCouponRequest {
  code: string;
  description?: string;
  discount_kind: 'PERCENTAGE';
  discount_value: number;
  max_redeems: number;
  expires_at?: string;
  is_active?: boolean;
}

interface CreateCouponResponse {
  success: boolean;
  data: {
    coupon: {
      code: string;
      description: string;
      discount_kind: string;
      discount_value: number;
      max_redeems: number;
      status: string;
      created_at: string;
      updated_at: string;
    };
  };
  message: string;
}

async function createCoupon(): Promise<CreateCouponResponse> {
  const request: CreateCouponRequest = {
    code: 'WELCOME12',
    description: 'Welcome discount 10% off',
    discount_kind: 'PERCENTAGE',
    discount_value: 12,
    max_redeems: 2,
    expires_at: '2024-12-31T23:59:59Z',
    is_active: true
  };

  const response = await fetch('https://api.monkei.co/api/v1/coupons', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-API-Key': 'YOUR_API_KEY'
    },
    body: JSON.stringify(request)
  });

  }

  return await response.json();
}

import (

    DiscountKind string `json:"discount_kind"`
    DiscountValue int   `json:"discount_value"`
    MaxRedeems   int   `json:"max_redeems"`
    ExpiresAt    string `json:"expires_at,omitempty"`
    IsActive     bool   `json:"is_active,omitempty"`

type CreateCouponResponse struct {
    Success bool `json:"success"`
    Data    struct {
        Coupon struct {
            Code         string `json:"code"`
            Description  string `json:"description"`
            DiscountKind string `json:"discount_kind"`
            DiscountValue int   `json:"discount_value"`
            MaxRedeems   int   `json:"max_redeems"`
            Status       string `json:"status"`
            CreatedAt    string `json:"created_at"`
            UpdatedAt    string `json:"updated_at"`
        } `json:"coupon"`
    } `json:"data"`
    Message string `json:"message"`
}

func main() {
    url := "https://api.monkei.co/api/v1/coupons"
    
    requestBody := CreateCouponRequest{
        Code:         "WELCOME12",
        Description:  "Welcome discount 10% off",
        DiscountKind: "PERCENTAGE",
        DiscountValue: 12,
        MaxRedeems:   2,
        ExpiresAt:    "2024-12-31T23:59:59Z",
        IsActive:     true,
    }
    
    jsonData, err := json.Marshal(requestBody)
    if err != nil {
        fmt.Printf("Error marshaling JSON: %v\n", err)
        return
    }
    
    req, err := http.NewRequest("POST", url, bytes.NewBuffer(jsonData))
    if err != nil {
        fmt.Printf("Error creating request: %v\n", err)
        return
    }
    
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("X-API-Key", "YOUR_API_KEY")
    
    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        fmt.Printf("Error making request: %v\n", err)
        return
    }
    defer resp.Body.Close()
    
    body, err := io.ReadAll(resp.Body)
    if err != nil {
        fmt.Printf("Error reading response: %v\n", err)
        return
    }
    
    var couponResponse CreateCouponResponse
    if err := json.Unmarshal(body, &couponResponse); err != nil {
        fmt.Printf("Error unmarshaling response: %v\n", err)
        return
    }
    
    fmt.Printf("Response: %+v\n", couponResponse)
