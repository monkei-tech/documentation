---
title: "Create Billing"
openapi: "POST /api/v1/bills"
description: "Create a new billing charge for a customer."
---

<ParamField body="frequency" type="string" required>
  The frequency of the billing.
  Allowed values: `ONE_TIME`, `MULTIPLE_PAYMENTS`.
</ParamField>

<ParamField body="customer_data" type="object" required>
  The customer information.
  
  <Note>
    When `frequency` is `MULTIPLE_PAYMENTS`, this field is **not required** and can be omitted.
  </Note>
  
  <Expandable title="properties">
    <ParamField body="customer_id" type="string">
      Existing customer ID (optional).
    </ParamField>
    <ParamField body="customer_metadata" type="object">
      Customer metadata for creating a new customer.
      <Expandable title="properties">
        <ParamField body="name" type="string">Customer name.</ParamField>
        <ParamField body="phone" type="string">Customer phone.</ParamField>
        <ParamField body="email" type="string">Customer email.</ParamField>
        <ParamField body="tax_id" type="string">Customer tax ID (CPF/CNPJ).</ParamField>
      </Expandable>
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="payment_methods" type="array" required>
  List of allowed payment methods.
  Allowed values: `PIX`, `BITCOIN`, `USDT`.
</ParamField>

<ParamField body="metadata" type="object">
  Additional metadata for the billing. This metadata will be included in webhook events sent to your application.
</ParamField>

<ParamField body="products" type="array" required>
  List of products or services. Product information will be included in webhook events sent to your application.
  <Expandable title="properties">
    <ParamField body="external_id" type="string" required>
      External product ID from your company's system. This ID also serves for idempotency - using the same `external_id` will return the same product.
    </ParamField>
    <ParamField body="name" type="string" required>Product name.</ParamField>
    <ParamField body="description" type="string">Product description.</ParamField>
    <ParamField body="price" type="integer" required>Price in cents.</ParamField>
    <ParamField body="quantity" type="integer" required>Quantity.</ParamField>
    <ParamField body="category" type="string">Product category.</ParamField>
  </Expandable>
</ParamField>

<ParamField body="return_url" type="string" required>
  URL to redirect when the user clicks the back button or cancels the payment.
</ParamField>

<ParamField body="completion_url" type="string" required>
  URL to redirect after payment completion.
</ParamField>

<ParamField body="allow_coupon" type="boolean">
  Whether to allow coupons.
</ParamField>

<ParamField body="coupons_allowed" type="array">
  List of allowed coupon codes.
</ParamField>
