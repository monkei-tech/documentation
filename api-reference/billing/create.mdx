---
title: "Create Billing"
openapi: "POST /api/v1/bills"
description: "Create a new billing charge for a customer."
---

<ParamField body="currency" type="string">
  Currency code for the billing. Defaults to `BRL` if not provided.
  Allowed values: `BRL`, `USD`.
</ParamField>

<ParamField body="idiom" type="string">
  Language/idiom for the payment page. Defaults to `pt-BR` if not provided.
  Allowed values: `pt-BR`, `en-US`.
</ParamField>

<ParamField body="frequency" type="string" required>
  The frequency of the billing.
  Allowed values: `ONE_TIME`, `MULTIPLE_PAYMENTS`.
</ParamField>

<ParamField body="customer_data" type="object" required>
  The customer information.
  
  <Note>
    When `frequency` is `MULTIPLE_PAYMENTS`, this field is **not required** and can be omitted.
  </Note>
  
  <Expandable title="properties">
    <ParamField body="customer_id" type="string">
      Existing customer ID (optional).
    </ParamField>
    <ParamField body="customer_metadata" type="object">
      Customer metadata for creating a new customer.
      <Expandable title="properties">
        <ParamField body="name" type="string">Customer name.</ParamField>
        <ParamField body="phone" type="string">Customer phone.</ParamField>
        <ParamField body="email" type="string">Customer email.</ParamField>
        <ParamField body="tax_id" type="string">Customer tax ID (CPF/CNPJ).</ParamField>
      </Expandable>
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="payment_methods" type="array" required>
  List of allowed payment methods.
  Allowed values: `PIX`, `BTC`, `USDT`.
</ParamField>

<ParamField body="metadata" type="object">
  Additional metadata for the billing. This metadata will be included in webhook events sent to your application.
</ParamField>

<ParamField body="products" type="array" required>
  List of products or services. Product information will be included in webhook events sent to your application.
  <Expandable title="properties">
    <ParamField body="external_id" type="string" required>
      External product ID from your company's system. This ID also serves for idempotency - using the same `external_id` will return the same product.
    </ParamField>
    <ParamField body="name" type="string" required>Product name.</ParamField>
    <ParamField body="description" type="string">Product description.</ParamField>
    <ParamField body="price" type="integer" required>Price in smallest unit (cents). Minimum value: 700 cents (R$ 7.00 or $7.00).</ParamField>
    <ParamField body="quantity" type="integer" required>Quantity.</ParamField>
    <ParamField body="category" type="string">Product category.</ParamField>
  </Expandable>
</ParamField>

<ParamField body="return_url" type="string" required>
  URL to redirect when the user clicks the back button or cancels the payment.
</ParamField>

<ParamField body="completion_url" type="string" required>
  URL to redirect after payment completion.
</ParamField>

<ParamField body="allow_coupon" type="boolean">
  Whether to allow coupons.
</ParamField>

<ParamField body="coupons_allowed" type="array">
  List of allowed coupon codes.
</ParamField>

```typescript TypeScript
interface CustomerMetadata {
  name: string;
  phone: string;
  email: string;
  tax_id: string;
}

interface CustomerData {
  customer_id?: string;
  customer_metadata?: CustomerMetadata;
}

interface Product {
  external_id: string;
  name: string;
  description?: string;
  price: number;
  quantity: number;
  category?: string;
}

interface CreateBillingRequest {
  currency?: 'BRL' | 'USD';
  idiom?: 'pt-BR' | 'en-US';
  frequency: 'ONE_TIME' | 'MULTIPLE_PAYMENTS';
  customer_data?: CustomerData;
  payment_methods: ('PIX' | 'BTC' | 'USDT')[];
  metadata?: Record<string, any>;
  products: Product[];
  return_url: string;
  completion_url: string;
  allow_coupon?: boolean;
  coupons_allowed?: string[];
}

async function createBilling(): Promise<any> {
  const request: CreateBillingRequest = {
    frequency: 'ONE_TIME',
    customer_data: {
      customer_id: '12345678901',
      customer_metadata: {
        name: 'John Doe',
        phone: '5511999999999',
        email: 'customer@example.com',
        tax_id: '12345678901'
      }
    },
    payment_methods: ['PIX', 'BTC', 'USDT'],
    metadata: { order_id: '123', customer_id: '456', source: 'ecommerce' },
    products: [
      {
        external_id: 'prod_test_001',
        name: 'Test Product',
        description: 'Detailed product description',
        price: 10050,
        quantity: 1,
        category: 'electronics'
      }
    ],
    return_url: 'https://example.com/return',
    completion_url: 'https://example.com/completion',
    allow_coupon: false,
    coupons_allowed: []
  };

  const response = await fetch('https://api.monkei.co/api/v1/bills', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-API-Key': 'YOUR_API_KEY'
    },
    body: JSON.stringify(request)
  });

  if (!response.ok) {
    throw new Error(`HTTP error! status: ${response.status}`);
  }

  return await response.json();
}
```

```go Go
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
)

type CustomerMetadata struct {
    Name  string `json:"name"`
    Phone string `json:"phone"`
    Email string `json:"email"`
    TaxID string `json:"tax_id"`
}

type CustomerData struct {
    CustomerID     string           `json:"customer_id,omitempty"`
    CustomerMetadata *CustomerMetadata `json:"customer_metadata,omitempty"`
}

type Product struct {
    ExternalID string `json:"external_id"`
    Name       string `json:"name"`
    Description string `json:"description,omitempty"`
    Price      int    `json:"price"`
    Quantity   int    `json:"quantity"`
    Category   string `json:"category,omitempty"`
}

type CreateBillingRequest struct {
    Currency       string        `json:"currency,omitempty"`
    Idiom          string        `json:"idiom,omitempty"`
    Frequency      string        `json:"frequency"`
    CustomerData   *CustomerData `json:"customer_data,omitempty"`
    PaymentMethods []string      `json:"payment_methods"`
    Metadata       map[string]interface{} `json:"metadata,omitempty"`
    Products       []Product     `json:"products"`
    ReturnURL      string        `json:"return_url"`
    CompletionURL  string        `json:"completion_url"`
    AllowCoupon    bool          `json:"allow_coupon,omitempty"`
    CouponsAllowed []string      `json:"coupons_allowed,omitempty"`
}

func main() {
    url := "https://api.monkei.co/api/v1/bills"
    
    requestBody := CreateBillingRequest{
        Currency: "BRL",
        Idiom: "pt-BR",
        Frequency: "ONE_TIME",
        CustomerData: &CustomerData{
            CustomerID: "12345678901",
            CustomerMetadata: &CustomerMetadata{
                Name:  "John Doe",
                Phone: "5511999999999",
                Email: "customer@example.com",
                TaxID: "12345678901",
            },
        },
        PaymentMethods: []string{"PIX", "BTC", "USDT"},
        Metadata: map[string]interface{}{
            "order_id":    "123",
            "customer_id": "456",
            "source":      "ecommerce",
        },
        Products: []Product{
            {
                ExternalID: "prod_test_001",
                Name:       "Test Product",
                Description: "Detailed product description",
                Price:      10050,
                Quantity:   1,
                Category:   "electronics",
            },
        },
        ReturnURL:     "https://example.com/return",
        CompletionURL: "https://example.com/completion",
        AllowCoupon:   false,
        CouponsAllowed: []string{},
    }
    
    jsonData, err := json.Marshal(requestBody)
    if err != nil {
        fmt.Printf("Error marshaling JSON: %v\n", err)
        return
    }
    
    req, err := http.NewRequest("POST", url, bytes.NewBuffer(jsonData))
    if err != nil {
        fmt.Printf("Error creating request: %v\n", err)
        return
    }
    
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("X-API-Key", "YOUR_API_KEY")
    
    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        fmt.Printf("Error making request: %v\n", err)
        return
    }
    defer resp.Body.Close()
    
    body, err := io.ReadAll(resp.Body)
    if err != nil {
        fmt.Printf("Error reading response: %v\n", err)
        return
    }
    
    var result map[string]interface{}
    if err := json.Unmarshal(body, &result); err != nil {
        fmt.Printf("Error unmarshaling response: %v\n", err)
        return
    }
    
    fmt.Printf("Response: %+v\n", result)
}
```
