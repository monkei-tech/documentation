---
title: "Account Balance"
openapi: "GET /api/v1/accounts/balance"
description: "Retrieve account balance information for the authenticated account"
---

Retrieve the current account balance information including balances for BRL, BTC, and USDT, along with account status.

## Endpoint

```http
GET /api/v1/accounts/balance
```

## Query Parameters

<ParamField query="refresh" type="boolean" default="false">
  Force refresh the cache. When set to `true`, the balance will be recalculated from the database instead of using cached values.
</ParamField>

<RequestExample>

```bash cURL
curl --request GET \
  --url 'https://api.monkei.co/api/v1/accounts/balance' \
  --header 'X-API-Key: YOUR_API_KEY'
```

```javascript JavaScript
const options = {
  method: 'GET',
  headers: {
    'X-API-Key': 'YOUR_API_KEY'
  }
};

fetch('https://api.monkei.co/api/v1/accounts/balance', options)
  .then(response => response.json())
  .then(response => console.log(response))
  .catch(err => console.error(err));
```

</RequestExample>

## Response Fields

| Field | Type | Description |
|-------|------|-------------|
| `accountNumber` | string | Unique account number identifier |
| `isActive` | boolean | Whether the account is currently active |
| `balances` | object | Account balances by currency/ledger type |

### Balances Object

The `balances` object contains balance information for each supported currency. Each currency has the following structure:

| Field | Type | Description |
|-------|------|-------------|
| `total` | string | Total balance (sum of available + hold) |
| `available` | string | Available balance that can be used for transactions |
| `hold` | string | Balance currently on hold (reserved for pending transactions) |

### Supported Currencies

The balances object may contain the following currency keys:

- `BRL`: Brazilian Reais balance
- `BTC`: Bitcoin balance
- `USDT`: Tether (USDT) balance

<Note>
  If a currency has no balance (all values are "0"), it may still appear in the response. The `total` field is calculated as the sum of `available` and `hold` balances.
</Note>

<Note>
  Balance values are returned as strings to maintain precision for decimal values, especially for cryptocurrencies.
</Note>

## Status Codes

| Code | Description |
|------|-------------|
| `200` | Account balance retrieved successfully |
| `401` | Unauthorized - invalid or missing API key |
| `404` | Not found - account not found |
| `500` | Internal server error |

## Examples

### Basic Balance Check

```javascript
async function getAccountBalance(apiKey) {
  const response = await fetch(
    'https://api.monkei.co/api/v1/accounts/balance',
    {
      method: 'GET',
      headers: {
        'X-API-Key': apiKey
      }
    }
  );

  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.message || 'Failed to get account balance');
  }

  const data = await response.json();
  return data.data;
}

// Usage
const balance = await getAccountBalance('YOUR_API_KEY');
console.log('Account Number:', balance.accountNumber);
console.log('Is Active:', balance.isActive);
console.log('BRL Balance:', balance.balances.BRL);
console.log('BTC Balance:', balance.balances.BTC);
console.log('USDT Balance:', balance.balances.USDT);
```

### Get Available Balance for a Currency

```javascript
async function getAvailableBalance(apiKey, currency = 'BRL') {
  const balance = await getAccountBalance(apiKey);
  const currencyBalance = balance.balances[currency];
  
  if (!currencyBalance) {
    return '0';
  }
  
  return currencyBalance.available;
}

// Usage
const availableBRL = await getAvailableBalance('YOUR_API_KEY', 'BRL');
const availableBTC = await getAvailableBalance('YOUR_API_KEY', 'BTC');
console.log('Available BRL:', availableBRL);
console.log('Available BTC:', availableBTC);
```

### Check if Account Has Sufficient Balance

```javascript
async function hasSufficientBalance(apiKey, currency, requiredAmount) {
  const balance = await getAccountBalance(apiKey);
  const currencyBalance = balance.balances[currency];
  
  if (!currencyBalance) {
    return false;
  }
  
  const available = parseFloat(currencyBalance.available);
  const required = parseFloat(requiredAmount);
  
  return available >= required;
}

// Usage
const hasEnough = await hasSufficientBalance('YOUR_API_KEY', 'BRL', '100.50');
if (hasEnough) {
  console.log('Account has sufficient balance');
} else {
  console.log('Insufficient balance');
}
```

### Get Total Balance Across All Currencies

```javascript
async function getTotalBalances(apiKey) {
  const balance = await getAccountBalance(apiKey);
  const totals = {};
  
  for (const [currency, currencyBalance] of Object.entries(balance.balances)) {
    totals[currency] = {
      total: currencyBalance.total,
      available: currencyBalance.available,
      hold: currencyBalance.hold
    };
  }
  
  return totals;
}

// Usage
const totals = await getTotalBalances('YOUR_API_KEY');
console.log('All balances:', totals);
```

### Python Example

```python
import requests

def get_account_balance(api_key, refresh=False):
    url = "https://api.monkei.co/api/v1/accounts/balance"
    headers = {
        'X-API-Key': api_key
    }
    params = {}
    
    if refresh:
        params['refresh'] = 'true'
    
    response = requests.get(url, headers=headers, params=params)
    response.raise_for_status()
    
    data = response.json()
    return data['data']

def get_available_balance(api_key, currency='BRL'):
    balance = get_account_balance(api_key)
    currency_balance = balance['balances'].get(currency)
    
    if not currency_balance:
        return '0'
    
    return currency_balance['available']

def has_sufficient_balance(api_key, currency, required_amount):
    balance = get_account_balance(api_key)
    currency_balance = balance['balances'].get(currency)
    
    if not currency_balance:
        return False
    
    available = float(currency_balance['available'])
    required = float(required_amount)
    
    return available >= required

# Usage
balance = get_account_balance('YOUR_API_KEY')
print(f"Account Number: {balance['accountNumber']}")
print(f"Is Active: {balance['isActive']}")
print(f"BRL Available: {get_available_balance('YOUR_API_KEY', 'BRL')}")
print(f"BTC Available: {get_available_balance('YOUR_API_KEY', 'BTC')}")

if has_sufficient_balance('YOUR_API_KEY', 'BRL', '100.50'):
    print('Account has sufficient balance')
else:
    print('Insufficient balance')
```

## Use Cases

### Monitor Account Balance

```javascript
async function monitorBalance(apiKey, currency, threshold, callback) {
  const balance = await getAccountBalance(apiKey);
  const currencyBalance = balance.balances[currency];
  
  if (!currencyBalance) {
    return;
  }
  
  const available = parseFloat(currencyBalance.available);
  
  if (available < threshold) {
    callback({
      currency,
      available,
      threshold,
      message: `Balance below threshold: ${available} < ${threshold}`
    });
  }
}

// Usage
monitorBalance('YOUR_API_KEY', 'BRL', 100, (alert) => {
  console.warn('Balance alert:', alert);
});
```

### Calculate Total Portfolio Value

```javascript
async function getPortfolioValue(apiKey, exchangeRates) {
  const balance = await getAccountBalance(apiKey);
  let totalValue = 0;
  
  // Assuming exchangeRates contains { BRL: 1, BTC: 250000, USDT: 5.2 }
  for (const [currency, currencyBalance] of Object.entries(balance.balances)) {
    const rate = exchangeRates[currency] || 1;
    const total = parseFloat(currencyBalance.total);
    totalValue += total * rate;
  }
  
  return totalValue;
}

// Usage
const exchangeRates = {
  BRL: 1,
  BTC: 250000,
  USDT: 5.2
};

const portfolioValue = await getPortfolioValue('YOUR_API_KEY', exchangeRates);
console.log(`Total portfolio value: BRL ${portfolioValue.toFixed(2)}`);
```

<Note>
  The balance information is cached by default to improve performance. Use the `refresh=true` query parameter when you need the most up-to-date balance information, such as after completing a transaction.
</Note>

<Warning>
  Balance values are returned as strings to maintain precision. When performing calculations, make sure to parse them as decimal numbers using appropriate libraries (e.g., `decimal.js` for JavaScript or `decimal.Decimal` for Python) to avoid floating-point precision issues.
</Warning>
